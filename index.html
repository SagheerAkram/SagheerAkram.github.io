// Function to render the main section structure export function renderSection(title, icon) { const section = document.createElement('section'); section.className = 'py-10'; section.innerHTML = `

${icon} ${title}

`; return section; } // Universal Card Component for all content types export function CardComponent(data, type) { let title, subtitle, body, footerTags, action; if (type === 'project') { title = data.name; subtitle = data.type || 'Project'; body = data.desc; footerTags = data.tags; action = ``; } else if (type === 'experience') { title = data.role; subtitle = `${data.company} (${data.period})`; body = data.desc; footerTags = []; } else if (type === 'education') { title = data.school; subtitle = data.degree; body = `Years: ${data.years}`; footerTags = []; } else if (type === 'article') { title = data.title; subtitle = data.date; body = data.body.substring(0, 100) + '...'; footerTags = data.tags; action = ``; } // Tailwind CSS for the component look and feel return `

${title}

${subtitle}

${body}

${footerTags && footerTags.length > 0 ? `
${footerTags.map(tag => `${tag}`).join('')}
` : '' } ${action ? action : ''}
`; }
